cmake_minimum_required(VERSION 3.10)
project(sigmarl_gazebo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gazebo 11 REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs nav_msgs sensor_msgs)
include_directories(${GAZEBO_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${GAZEBO_CXX_FLAGS}")
foreach(plugin drive camera)
  add_library(sigmarl_${plugin} SHARED plugins/${plugin}.cpp)
  target_link_libraries(sigmarl_${plugin} ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
  target_compile_options(sigmarl_${plugin} PRIVATE -Wall -Wextra)
endforeach()
